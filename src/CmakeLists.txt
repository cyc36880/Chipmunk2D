cmake_minimum_required(VERSION 3.10)
project(main)

# Chipmunk2D 头文件
include_directories(${CMAKE_CURRENT_SOURCE_DIR}
    ${CMAKE_CURRENT_SOURCE_DIR}/components/chipmunk2D/include)

# SDL2 配置 (MinGW / msys64)
set(SDL2_INCLUDE_DIR D:/msys64/mingw64/include/SDL2)
set(SDL2_LIBRARY_DIR D:/msys64/mingw64/lib)

# Chipmunk2D 静态库配置
set(BUILD_STATIC ON)
add_subdirectory(components/chipmunk2D/src)

# 创建可执行文件
add_executable(main)

target_include_directories(main PRIVATE ${SDL2_INCLUDE_DIR}) 
target_link_directories(main PRIVATE ${SDL2_LIBRARY_DIR})
target_sources(
    main 
    PRIVATE 
        main.c
        # ${CHIPMUNK_SOURCES}
)

# 链接 SDL2 
target_link_libraries(main PRIVATE SDL2 chipmunk_static)
